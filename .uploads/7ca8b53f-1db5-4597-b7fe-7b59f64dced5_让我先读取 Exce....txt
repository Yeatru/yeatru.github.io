```
function calc(sku, mainCategory, qty, countryCode) {
  const unitKg = estimateWeight(sku, mainCategory);
  const totalKg = unitKg * qty;
  const rate = RATES_BY_COUNTRY[countryCode];
  if (!rate) return null;
  // Per-package weight limit (country-specific)
  const maxPerPkg = rate.max_kg;
  const numPackages = Math.max(1, Math.ceil(totalKg / maxPerPkg));
  // Split evenly, each rounded up to next 0.5kg, min 0.5kg
  const perPkgRaw = totalKg / numPackages;
  const roundUp05 = (x) => Math.max(0.5, Math.ceil(x * 2) / 2);
  const perPgBillable = roundUp05(perPkgRaw);
  const totalBillableKg = perPkgBillable * numPackages;
  const freight = totalBillableKg * rate.rate_per_kg_usd;
  const perPkgFee = numPackages * rate.per_package_usd;
  const total = freight + perPkgFee;
  return { unitKg, totalKg, numPackages, perPkgBillable, totalBillableKg, freight, perPkgFee, total, eta: rate.eta, notes: rate.notes };
}
```
